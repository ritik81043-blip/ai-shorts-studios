        function renderVideo() {
            const consoleDiv = document.getElementById('render-console');
            const bar = document.getElementById('render-bar');
            const percent = document.getElementById('render-percent');
            const statusText = document.getElementById('render-status-text');
            
            consoleDiv.classList.remove('hidden');
            let progress = 0;
            
            const interval = setInterval(() => {
                progress += 10;
                bar.style.width = progress + '%';
                percent.innerText = progress + '%';
                
                if(progress === 30) statusText.innerText = 'Synthesizing ElevenLabs Voiceover & SFX...';
                if(progress === 60) statusText.innerText = 'Compositing PNG Cutout Stickers & Keyframes...';
                if(progress === 90) statusText.innerText = 'Remotion & FFmpeg H.264 MP4 Export...';
                
                if(progress >= 100) {
                    clearInterval(interval);
                    statusText.innerText = '✨ Render Complete! MP4 Ready for Download.';
                    bar.classList.add('bg-emerald-500');
                }
            }, 300);
        }
